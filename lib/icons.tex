%% lib/icons.tex — Platform icon registry.
%%
%% Edit this file to swap any platform glyph. The interface is one line per
%% platform:
%%
%%     \definePIcon{<name>}{<glyph macro>}
%%
%% Glyph macros come from Font Awesome 5, Devicon, and Simple Icons.
%% External font codepoints live in icon-glyphs.tex; default public mappings
%% live in icon-registry.tex. Keep this file as the stable import facade.
%%
%% Sizing & color come from \PIcon — do not embed those in your mapping;
%% just give the glyph itself.
%%
%% Example overrides:
%%   \definePIcon{blog}{\faRssSquare}                % Font Awesome RSS square
%%   \definePIcon{go}{\DeviconGo}                    % Devicon Go glyph
%%   \definePIcon{etcd}{\SimpleIconEtcd}             % Simple Icons etcd glyph
%%   \definePIcon{xiaohongshu}{\TextPIcon{XHS}}      % compact text badge

%% ----------------------------------------------------------------------
%% Internal storage + lookup
%% ----------------------------------------------------------------------
\makeatletter
\newcount\inkumo@savedlostchars
\inkumo@savedlostchars=\tracinglostchars
\tracinglostchars=0
\newif\ifinkumo@devicon
\IfFileExists{./assets/devicon/devicon.ttf}{%
  \inkumo@devicontrue
  \newfontface\inkumoDevicon{devicon.ttf}[Path=./assets/devicon/]%
}{%
  \inkumo@deviconfalse
}
\newif\ifinkumo@simpleicons
\IfFileExists{./assets/simple-icons/SimpleIcons.ttf}{%
  \inkumo@simpleiconstrue
  \newfontface\inkumoSimpleIcons{SimpleIcons.ttf}[Path=./assets/simple-icons/]%
}{%
  \inkumo@simpleiconsfalse
}
\tracinglostchars=\inkumo@savedlostchars
\NewDocumentCommand{\definePIcon}{m m}{%
  \expandafter\def\csname inkumo@picon@#1\endcsname{#2}%
}
\NewDocumentCommand{\DIcon}{m}{%
  \ifinkumo@devicon%
    \raisebox{-0.04ex}{{\inkumoDevicon\symbol{"#1}}}%
  \else%
    \faQuestionCircle
  \fi%
}
\NewDocumentCommand{\SIcon}{m}{%
  \ifinkumo@simpleicons%
    {\tracinglostchars=0\raisebox{-0.04ex}{{\inkumoSimpleIcons\symbol{"#1}}}}%
  \else%
    \faQuestionCircle
  \fi%
}
\NewDocumentCommand{\SIconCompact}{m}{%
  \ifinkumo@simpleicons%
    {\tracinglostchars=0\raisebox{-0.04ex}{\scalebox{0.95}{{\inkumoSimpleIcons\symbol{"#1}}}}}%
  \else%
    \faQuestionCircle
  \fi%
}
\NewDocumentCommand{\TextPIcon}{m}{%
  \tikz[baseline=(x.base)]{%
    \node[
      fill=kami@brandtint,
      draw=kami@brandlight!18!kami@border,
      line width=0.2pt,
      rounded corners=0.8pt,
      inner xsep=1.45pt,
      inner ysep=0.4pt,
      text=kami@brand,
      font={\fontsize{5.4pt}{5.4pt}\selectfont\bfseries},
      outer sep=0pt,
    ] (x) {#1};%
  }%
}
\NewDocumentCommand{\PIcon}{m}{%
  \begingroup
  \color{kami@darkwarm}%
  \fontsize{8.8pt}{8.8pt}\selectfont
  \ifcsname inkumo@picon@#1\endcsname
    \csname inkumo@picon@#1\endcsname
  \else
    \faQuestionCircle
  \fi
  \endgroup
}
\makeatother

\input{lib/icon-registry.tex}
