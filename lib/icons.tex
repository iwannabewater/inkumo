%% lib/icons.tex — Platform icon registry.
%%
%% Edit this file to swap any platform glyph. The interface is one line per
%% platform:
%%
%%     \definePIcon{<name>}{<glyph macro>}
%%
%% Glyph macros come from the fontawesome5 package (default backend) — see
%% https://ctan.org/pkg/fontawesome5 for the full list. You can also define
%% your own custom glyph (e.g. inline TeX, an image, an SVG) and bind it to a
%% name; the rest of the resume will pick it up wherever \PIcon{<name>} is
%% used.
%%
%% Sizing & color come from \PIcon — do not embed those in your mapping;
%% just give the glyph itself.
%%
%% Example overrides:
%%   \definePIcon{wechat}{\textbf{\textsc{wx}}}      % literal text
%%   \definePIcon{blog}{\faRssSquare}                % FA RSS square
%%   \definePIcon{xiaohongshu}{\textbf{XHS}}         % no FA glyph yet — use text

%% ----------------------------------------------------------------------
%% Internal storage + lookup
%% ----------------------------------------------------------------------
\makeatletter
\NewDocumentCommand{\definePIcon}{m m}{%
  \expandafter\def\csname inkumo@picon@#1\endcsname{#2}%
}
\NewDocumentCommand{\PIcon}{m}{%
  \begingroup
  \color{kami@darkwarm}%
  \fontsize{8.8pt}{8.8pt}\selectfont
  \ifcsname inkumo@picon@#1\endcsname
    \csname inkumo@picon@#1\endcsname
  \else
    \faQuestionCircle
  \fi
  \endgroup
}
\makeatother
