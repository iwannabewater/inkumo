%% lib/icons.tex — Platform icon registry.
%%
%% Edit this file to swap any platform glyph. The interface is one line per
%% platform:
%%
%%     \definePIcon{<name>}{<glyph macro>}
%%
%% Glyph macros come from Font Awesome 5, Devicon, and Simple Icons.
%% External font codepoints live in icon-glyphs.tex; default public mappings
%% live in icon-registry.tex. Keep this file as the stable import facade.
%%
%% Geometry and color come from the single \PIcon component. Provider adapters
%% only select a font and return a raw glyph; do not size or position mappings.
%%
%% Example overrides:
%%   \definePIcon{blog}{\faRssSquare}                % Font Awesome RSS square
%%   \definePIcon{go}{\DeviconGo}                    % Devicon Go glyph
%%   \definePIcon{etcd}{\SimpleIconEtcd}             % Simple Icons etcd glyph
%%   \definePIcon{xiaohongshu}{\TextPIcon{XHS}}      % compact text badge

%% ----------------------------------------------------------------------
%% Internal storage + lookup
%% ----------------------------------------------------------------------

\newsavebox{\inkumo@iconrawbox}
\newsavebox{\inkumo@iconfitbox}
% Preserve each glyph's natural proportions. The component only caps outliers,
% gives narrow icons a stable minimum slot, and aligns providers to one baseline.
\newlength{\inkumo@iconmaxwidth}
\newlength{\inkumo@iconmaxheight}
\newlength{\inkumo@icontallmaxheight}
\newlength{\inkumo@iconminslotwidth}
\newlength{\inkumo@iconslotheight}
\newlength{\inkumo@iconslotdepth}
\setlength{\inkumo@iconmaxwidth}{11pt}
\setlength{\inkumo@iconmaxheight}{8.2pt}
\setlength{\inkumo@icontallmaxheight}{8.6pt}
\setlength{\inkumo@iconminslotwidth}{8.8pt}
\setlength{\inkumo@iconslotheight}{7.55pt}
\setlength{\inkumo@iconslotdepth}{1.05pt}
\newcommand{\inkumo@fiticon}[1]{%
  \sbox{\inkumo@iconrawbox}{#1}%
  \sbox{\inkumo@iconfitbox}{\usebox{\inkumo@iconrawbox}}%
  \dimen@=\ht\inkumo@iconrawbox
  \advance\dimen@ by \dp\inkumo@iconrawbox
  \@tempdima=\dimen@
  \multiply\@tempdima by 4
  \divide\@tempdima by 5
  \ifdim\wd\inkumo@iconrawbox<\@tempdima
    \@tempdimb=\inkumo@icontallmaxheight
  \else
    \@tempdimb=\inkumo@iconmaxheight
  \fi
  \ifdim\dimen@>\@tempdimb
    \sbox{\inkumo@iconfitbox}{%
      \resizebox*{!}{\@tempdimb}{\usebox{\inkumo@iconrawbox}}%
    }%
  \fi
  \ifdim\wd\inkumo@iconfitbox>\inkumo@iconmaxwidth
    \sbox{\inkumo@iconfitbox}{%
      \resizebox{\inkumo@iconmaxwidth}{!}{\usebox{\inkumo@iconrawbox}}%
    }%
  \fi
}
\newcommand{\inkumo@iconcomponent}[1]{%
  \inkumo@fiticon{#1}%
  \dimen@=\inkumo@iconslotheight
  \advance\dimen@ by -\inkumo@iconslotdepth
  \divide\dimen@ by 2
  \@tempdima=\ht\inkumo@iconfitbox
  \advance\@tempdima by -\dp\inkumo@iconfitbox
  \divide\@tempdima by 2
  \advance\dimen@ by -\@tempdima
  \raisebox{\dimen@}[\inkumo@iconslotheight][\inkumo@iconslotdepth]{%
    \ifdim\wd\inkumo@iconfitbox<\inkumo@iconminslotwidth
      \makebox[\inkumo@iconminslotwidth][c]{\usebox{\inkumo@iconfitbox}}%
    \else
      \usebox{\inkumo@iconfitbox}%
    \fi
  }%
}

  \else

  \fi
  \endgroup
