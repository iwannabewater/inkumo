%% lib/icons.tex — Platform icon registry.
%%
%% Edit this file to swap any platform glyph. The interface is one line per
%% platform:
%%
%%     \definePIcon{<name>}{<glyph macro>}
%%
%% Glyph macros come from Font Awesome 5, Devicon, and Simple Icons.
%% External font codepoints live in icon-glyphs.tex; default public mappings
%% live in icon-registry.tex. Keep this file as the stable import facade.
%%
%% Geometry and color come from the single \PIcon component. Provider adapters
%% only select a font and return a raw glyph; do not size or position mappings.
%%
%% Example overrides:
%%   \definePIcon{blog}{\faRssSquare}                % Font Awesome RSS square
%%   \definePIcon{go}{\DeviconGo}                    % Devicon Go glyph
%%   \definePIcon{etcd}{\SimpleIconEtcd}             % Simple Icons etcd glyph
%%   \definePIcon{xiaohongshu}{\TextPIcon{XHS}}      % compact text badge

%% ----------------------------------------------------------------------
%% Internal storage + lookup
%% ----------------------------------------------------------------------
\makeatletter
\newcount\inkumo@savedlostchars
\inkumo@savedlostchars=\tracinglostchars
\tracinglostchars=0
\newif\ifinkumo@devicon
\IfFileExists{./assets/devicon/devicon.ttf}{%
  \inkumo@devicontrue
  \newfontface\inkumoDevicon{devicon.ttf}[Path=./assets/devicon/]%
}{%
  \inkumo@deviconfalse
}
\newif\ifinkumo@simpleicons
\IfFileExists{./assets/simple-icons/SimpleIcons.ttf}{%
  \inkumo@simpleiconstrue
  \newfontface\inkumoSimpleIcons{SimpleIcons.ttf}[Path=./assets/simple-icons/]%
}{%
  \inkumo@simpleiconsfalse
}
\tracinglostchars=\inkumo@savedlostchars
\NewDocumentCommand{\definePIcon}{m m}{%
  \expandafter\def\csname inkumo@picon@#1\endcsname{#2}%
}
\NewDocumentCommand{\DIcon}{m}{%
  \ifinkumo@devicon%
    {\inkumoDevicon\symbol{"#1}}%
  \else%
    \faQuestionCircle
  \fi%
}
\NewDocumentCommand{\SIcon}{m}{%
  \ifinkumo@simpleicons%
    {\tracinglostchars=0\inkumoSimpleIcons\symbol{"#1}}%
  \else%
    \faQuestionCircle
  \fi%
}
\NewDocumentCommand{\TextPIcon}{m}{%
  \tikz[baseline=(x.base)]{%
    \node[
      fill=kami@brandtint,
      draw=kami@brandlight!18!kami@border,
      line width=0.2pt,
      rounded corners=0.8pt,
      inner xsep=1.45pt,
      inner ysep=0.4pt,
      text=kami@brand,
      font={\fontsize{5.4pt}{5.4pt}\selectfont\bfseries},
      outer sep=0pt,
    ] (x) {#1};%
  }%
}
\newsavebox{\inkumo@iconrawbox}
\newsavebox{\inkumo@iconfitbox}
\newlength{\inkumo@iconvisualwidth}
\newlength{\inkumo@iconvisualheight}
\newlength{\inkumo@iconslotwidth}
\newlength{\inkumo@iconslotheight}
\newlength{\inkumo@iconslotdepth}
\setlength{\inkumo@iconvisualwidth}{8.2pt}
\setlength{\inkumo@iconvisualheight}{8.2pt}
\setlength{\inkumo@iconslotwidth}{8.8pt}
\setlength{\inkumo@iconslotheight}{7.25pt}
\setlength{\inkumo@iconslotdepth}{0.95pt}
\newcommand{\inkumo@iconcomponent}[1]{%
  \sbox{\inkumo@iconrawbox}{#1}%
  \sbox{\inkumo@iconfitbox}{%
    \resizebox{\inkumo@iconvisualwidth}{!}{\usebox{\inkumo@iconrawbox}}%
  }%
  \dimen@=\ht\inkumo@iconfitbox
  \advance\dimen@ by \dp\inkumo@iconfitbox
  \ifdim\dimen@>\inkumo@iconvisualheight
    \sbox{\inkumo@iconfitbox}{%
      \resizebox*{!}{\inkumo@iconvisualheight}{\usebox{\inkumo@iconrawbox}}%
    }%
  \fi
  \dimen@=\inkumo@iconslotheight
  \advance\dimen@ by -\inkumo@iconslotdepth
  \divide\dimen@ by 2
  \@tempdima=\ht\inkumo@iconfitbox
  \advance\@tempdima by -\dp\inkumo@iconfitbox
  \divide\@tempdima by 2
  \advance\dimen@ by -\@tempdima
  \raisebox{\dimen@}[\inkumo@iconslotheight][\inkumo@iconslotdepth]{%
    \makebox[\inkumo@iconslotwidth][c]{\usebox{\inkumo@iconfitbox}}%
  }%
}
\NewDocumentCommand{\PIcon}{O{kami@darkwarm} m}{%
  \begingroup
  \color{#1}%
  \fontsize{8.8pt}{8.8pt}\selectfont
  \ifcsname inkumo@picon@#2\endcsname
    \inkumo@iconcomponent{\csname inkumo@picon@#2\endcsname}%
  \else
    \inkumo@iconcomponent{\faQuestionCircle}%
  \fi
  \endgroup
}
\makeatother
